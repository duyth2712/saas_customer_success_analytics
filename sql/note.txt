--- invalid_date_flag
	CASE
	    WHEN TRY_CAST(subscription_start_date AS DATE) IS NULL
	      OR (subscription_end_date IS NOT NULL 
	          AND TRY_CAST(subscription_end_date AS DATE) 
	              < TRY_CAST(subscription_start_date AS DATE))
	      OR (TRY_CAST(last_login_date AS DATE) > CAST(GETDATE() AS DATE))
	    THEN 1 ELSE 0
	END AS invalid_date_flag,
	
	--- invalid_usage_flag
	CASE
	    WHEN TRY_CAST(monthly_active_users AS INT) 
	            > TRY_CAST(user_count AS INT)
	      OR TRY_CAST(feature_usage_score AS INT) < 0
	      OR TRY_CAST(feature_usage_score AS INT) > 1000
	    THEN 1 ELSE 0
	END AS invalid_usage_flag,
	
	--- invalid_retention_flag
	CASE
	    WHEN TRY_CAST(retention_rate_6m AS DECIMAL(5,2)) < 0
	      OR TRY_CAST(retention_rate_6m AS DECIMAL(5,2)) > 100
	      OR TRY_CAST(retention_rate_12m AS DECIMAL(5,2)) < 0
	      OR TRY_CAST(retention_rate_12m AS DECIMAL(5,2)) > 100
	      OR TRY_CAST(retention_rate_12m AS DECIMAL(5,2))
	            > TRY_CAST(retention_rate_6m AS DECIMAL(5,2))
	      OR TRY_CAST(churn_risk_score AS INT) NOT BETWEEN 1 AND 10
		THEN 1 ELSE 0
	END AS invalid_retention_flag,
